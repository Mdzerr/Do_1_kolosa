Klasa Segment — prywatne pola, kopie w konstruktorze, długość:
-------------------
public class Segment {
    private Point p1, p2;
    public Segment(Point a, Point b) { this.p1=new Point(a); this.p2=new Point(b);}
    public double length() {
        double dx = p2.getX() - p1.getX(), dy = p2.getY() - p1.getY();
        return Math.sqrt(dx*dx + dy*dy);
    }
    public Point getP1() { return p1; }
    public Point getP2() { return p2; }
    // Static — najdłuższy z tablicy
    public static Segment findLongest(Segment[] arr) {
        return Arrays.stream(arr).max(Comparator.comparingDouble(Segment::length)).orElse(null);
    }
    // Odcinek prostopadły, przecinający się w środku
    public Segment perpendicular() {
        double mx = (p1.getX()+p2.getX())/2, my = (p1.getY()+p2.getY())/2;
        double dx = p2.getX()-p1.getX(), dy = p2.getY()-p1.getY();
        double nx = -dy, ny = dx, len = Math.sqrt(nx*nx+ny*ny)/2;
        double ux = nx/Math.sqrt(nx*nx+ny*ny), uy = ny/Math.sqrt(nx*nx+ny*ny);
        Point q1 = new Point(mx+ux*len, my+uy*len);
        Point q2 = new Point(mx-ux*len, my-uy*len);
        return new Segment(q1, q2);
    }
}

Polygon — głęboka kopia, toSvg, boundingBox:

public class Polygon {
    private Point[] pts;
    public Polygon(Point[] pts) { this.pts = Arrays.stream(pts).map(Point::new).toArray(Point[]::new);}
    public Polygon(Polygon orig) { this(orig.pts); }
    public String toSvg() {
        String s = Arrays.stream(pts).map(p -> p.getX()+","+p.getY()).collect(Collectors.joining(" "));
        return "<polygon points=\"" + s + "\" fill=\"none\" stroke=\"black\"/>";
    }
    // Bounding box — tablica Pointów
    public record BoundingBox(double x, double y, double w, double h) {}
    public BoundingBox boundingBox() {
        double minX=Arrays.stream(pts).mapToDouble(Point::getX).min().orElse(0);
        double maxX=Arrays.stream(pts).mapToDouble(Point::getX).max().orElse(0);
        double minY=Arrays.stream(pts).mapToDouble(Point::getY).min().orElse(0);
        double maxY=Arrays.stream(pts).mapToDouble(Point::getY).max().orElse(0);
        return new BoundingBox(minX,minY,maxX-minX,maxY-minY);
    }
}

import java.time.LocalDate;
public class Person implements Comparable<Person>, Serializable {
    private String firstName, lastName;
    private LocalDate birth, death;
    private Set<Person> children = new HashSet<>();
    public Person(String f, String l, LocalDate b, LocalDate d) { firstName=f; lastName=l; birth=b; death=d; }
    public boolean adopt(Person ch) { return children.add(ch);}
    public Person getYoungestChild() { return children.stream().max(Comparator.naturalOrder()).orElse(null);}
    @Override public int compareTo(Person o) { return this.birth.compareTo(o.birth); }
    public List<Person> getChildren() { List<Person> l=new ArrayList<>(children); l.sort(Comparator.naturalOrder()); return l;}
    public String getFirstName() {return firstName;}
    public String getLastName() {return lastName;}
    public LocalDate getBirth() {return birth;}
    public LocalDate getDeath() {return death;}
    public String toString() { return firstName+" "+lastName+" ("+birth+")";}

    // CSV: static fabryczna
    public static Person fromCsvLine(String line) throws NegativeLifespanException {
        String[] p = line.split(",");
        String[] names = p[0].trim().split("\\s+");
        String fname = names[0], lname = names.length>1?names[1]:"";
        LocalDate birth = LocalDate.parse(p[1].trim());
        LocalDate death = (p[2].trim().isEmpty()) ? null : LocalDate.parse(p[2].trim());
        if (death!=null && death.isBefore(birth)) throw new NegativeLifespanException(fname+" "+lname,birth,death);
        return new Person(fname,lname,birth,death);
    }
    // Serializacja
    public static void toBinaryFile(List<Person> persons, String path) throws IOException {
        try (ObjectOutputStream o=new ObjectOutputStream(new FileOutputStream(path))) {o.writeObject(persons);}
    }
    public static List<Person> fromBinaryFile(String path) throws IOException,ClassNotFoundException {
        try(ObjectInputStream i=new ObjectInputStream(new FileInputStream(path))){return (List<Person>)i.readObject();}
    }
}

    }

public class CustomList<T> extends AbstractList<T> {
    private static class Node<T> { T d; Node<T> n; Node(T d){this.d=d;} }
    private Node<T> head, tail; private int len=0;

    public void addLast(T v) { Node<T> n=new Node<>(v); if(tail==null) head=tail=n;
        else{tail.n=n;tail=n;} len++; }
    public void addFirst(T v) { Node<T> n=new Node<>(v); if(head==null)head=tail=n;
        else{n.n=head;head=n;} len++; }
    public T getFirst(){if(head==null)throw new NoSuchElementException(); return head.d;}
    public T getLast(){if(tail==null)throw new NoSuchElementException(); return tail.d;}
    public T removeFirst() { if(head==null)throw new NoSuchElementException();
        T d=head.d; head=head.n;if(head==null)tail=null;len--;return d;}
    public T removeLast() {
        if(tail==null)throw new NoSuchElementException();
        if(head==tail){T d=tail.d;head=tail=null;len--;return d;}
        Node<T> prev=head;while(prev.n!=tail)prev=prev.n;
        T d=tail.d; tail=prev;tail.n=null;len--;return d;}
    @Override public boolean add(T t){addLast(t);return true;}
    @Override public int size(){return len;}
    @Override public T get(int idx) {
        Node<T> cur=head;for(int i=0;i<idx;i++)cur=cur.n;
        if(cur==null)throw new IndexOutOfBoundsException(); return cur.d;}
    @Override public Iterator<T> iterator() {
        return new Iterator<T>() {
            Node<T> c=head;
            public boolean hasNext(){return c!=null;}
            public T next(){if(!hasNext())throw new NoSuchElementException();T d=c.d; c=c.n; return d;}
        };
    }
    @Override public Spliterator<T> spliterator(){ return Spliterators.spliterator(this,0);}
    public Stream<T> stream(){ return StreamSupport.stream(spliterator(),false);}
}

public class DeathCauseStatistic {
    private String icdCode;
    private int[] deathsByAge; // zgony w grupach wiekowych

    public DeathCauseStatistic(String icdCode, int[] deathsByAge) {
        this.icdCode = icdCode;
        this.deathsByAge = deathsByAge;
    }
    public static DeathCauseStatistic fromCsvLine(String line) {
        String[] parts = line.split("\t");
        String code = parts[0].trim();
        if (!code.matches("[A-Z]\\d{2}.*")) throw new IllegalArgumentException("Nieprawidłowy kod ICD: "+code);
        int[] deaths = Arrays.stream(Arrays.copyOfRange(parts,2,parts.length))
                             .mapToInt(s -> Integer.parseInt(s.trim())).toArray();
        return new DeathCauseStatistic(code, deaths);
    }
    public String getIcdCode(){return icdCode;}
    public int[] getDeathsByAge(){return deathsByAge.clone();}

    // Wejściowa: jaka grupa wiekowa, jakie zgony?
    public static class AgeBracketDeaths {
        public final int young, old, deathCount;
        public AgeBracketDeaths(int y, int o, int c){young=y; old=o; deathCount=c;}
    }
    private static final int[] AGE_LIMITS = {0, 10, 20, 30, 40, 50, 60, 70, 80};
    public AgeBracketDeaths getBracket(int age) {
        for(int i=0;i<AGE_LIMITS.length-1;i++) {
            if(age>=AGE_LIMITS[i]&&age<AGE_LIMITS[i+1]) return new AgeBracketDeaths(AGE_LIMITS[i],AGE_LIMITS[i+1]-1,deathsByAge[i]);
        }
        int i=deathsByAge.length-1;
        return new AgeBracketDeaths(AGE_LIMITS[i], Integer.MAX_VALUE, deathsByAge[i]);
    }
}

public interface ICDCodeTabular { String getDescription(String icdCode); }

public class ICDCodeTabularOptimizedForTime implements ICDCodeTabular {
    private final Map<String,String> map = new HashMap<>();
    public ICDCodeTabularOptimizedForTime(String filepath) throws IOException {
        List<String> lines = Files.readAllLines(Paths.get(filepath));
        for(String line: lines.subList(87, lines.size())){
            if(line.matches("[A-Z]\\d{2}.*\\s+.+")){
                String[] spl = line.split("\\s+",2);
                map.put(spl[0],spl[1]);
            }
        }
    }
    public String getDescription(String icdCode) {
        String desc = map.get(icdCode);
        if(desc==null)throw new IndexOutOfBoundsException("Brak kodu "+icdCode);
        return desc;
    }
}

public class ICDCodeTabularOptimizedForMemory implements ICDCodeTabular {
    private final String filepath;
    public ICDCodeTabularOptimizedForMemory(String filepath){this.filepath=filepath;}
    public String getDescription(String icdCode){
        try(Stream<String> lines = Files.lines(Paths.get(filepath))) {
            return lines.skip(87)
                .filter(l->l.startsWith(icdCode+" "))
                .map(l->l.substring(icdCode.length()).trim())
                .findFirst()
                .orElseThrow(() -> new IndexOutOfBoundsException(icdCode));
        }catch(IOException e){throw new UncheckedIOException(e);}
    }
}
